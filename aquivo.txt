#include <stdio.h>
#include <stdlib.h>

int main(void)
{
    // Função para abrir um arquivo 
    FILE * arq;
    arq = fopen("arquivo.txt", "w");
    if (arq == NULL){
        printf("Não e possivel criar o arquivo");
        exit(1);
    }
    else{
        printf("Arquivo criado");
    }

    fclose(arq);

    return 0;
}
